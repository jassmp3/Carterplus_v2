  }

        if (!refs.length) return;
        const chosen = new Set(sel.map((s) => s.el));
        const items = scanRegion(vs).filter((i) => i.cand && !chosen.has(i.el));
        let best = null, bd = Infinity;
        for (const it of items) for (const r of refs) {
          const d = Math.hypot(it.x - r.x, it.y - r.y);
          if (d < bd) { bd = d; best = it; }
        }
        const S = median(refs.map((r) => Math.max(r.w, r.h)));
        if (!best || bd > S * 2.6) { log({ orphan: 'no seat found' }); judgeAfterDialog(true); return; }
        const T = makeTargets([best]);
        captureAfter(T);
        fireTarget(T[0], false);
        lastT = lastT.concat(T);
        log({ selected: 'extra', why: 'single seat left open', n: lastT.length });
        await verifySelect(T, my, { reserved: false });
        if (my !== epoch) return;
        phase = 'reserve'; reserveAt = performance.now();
        reserveFast(my, { reserved: false }, performance.now());
      } finally { fixing = false; }
    }, 250);
  }

  const DLG = '[role=dialog],[role=alertdialog],[aria-modal="true"],dialog[open],[class*="modal" i],[class*="dialog" i],[class*="popup" i],[class*="overlay" i],[data-testid*="modal" i],[data-testid*="dialog" i],[data-testid*="alert" i]';
  const OK_TIERS = [/^agree\s*(&|and)\s*continue$/, /^continue$/, /^agree$/, /^accept$/, /^ok(ay)?$/, /^got it$/, /^confirm$/, /^yes(,.*)?$/, /^submit$/, /^proceed$/, /^understood$|^i understand$/, /^try again$|^retry$/, /^close$|^dismiss$/];
  const LOOSE = [/agree/, /continue/, /\bok\b/, /accept/, /confirm/, /submit/, /close|dismiss/];
  const NO = /^(cancel|no|no thanks|not now|back|go back|decline|exit|leave|x|\u00d7)$/;
  const ERR = /(went wrong|not available|unavailable|no longer available|error|failed|unable to|expired)/i;
  const norm = (s) => s.replace(/&amp;/g, '&').replace(/\s+/g, ' ').trim().toLowerCase();
  const clickedAt = new WeakMap();
  function findDialogs() {
    const out = [];
    for (const d of document.querySelectorAll(DLG)) {
      if (d.closest('[data-carter]') || !visible(d)) continue;
      const r = d.getBoundingClientRect();
      if (r.width < 120 || r.height < 60) continue;
      const aria = d.matches('[role=dialog],[role=alertdialog],[aria-modal="true"],dialog[open]');
      if (!aria) { const cs = getComputedStyle(d); if (!/fixed|absolute|sticky/.test(cs.position)) continue; }
      const btns = [];
      for (const b of d.querySelectorAll('button,[role=button],a[href],input[type=button],input[type=submit]')) {
        if (!visible(b)) continue;
        const text = norm(b.innerText || b.textContent || b.value || '');
        const label = norm(b.getAttribute('aria-label') || '');
        btns.push({ el: b, text: text || label, icon: !text && /close|dismiss/.test(label) });
      }
      if (btns.length) out.push({ el: d, btns, aria, text: (d.innerText || '').slice(0, 500) });
    }
    return out;
  }
  function chooseButton(btns, allowIcon, allowSingle) {
    let best = null, bi = 1e9;
    for (const b of btns) {
      if (NO.test(b.text) || !enabled(b.el)) continue;
      let i = OK_TIERS.findIndex((re) => re.test(b.text));
      if (i < 0) { const j = LOOSE.findIndex((re) => re.test(b.text)); if (j >= 0) i = OK_TIERS.length + j; }
      if (i < 0 && b.icon && allowIcon) i = OK_TIERS.length + LOOSE.length;
      if (i < 0 && allowSingle && btns.filter((x) => !NO.test(x.text)).length === 1 && b.text) i = OK_TIERS.length + LOOSE.length + 1;
      if (i >= 0 && i < bi) { bi = i; best = b; }
    }
    return best;
  }
  function handleDialogs() {
    if (!engineOn() || !rects.length) return;
    for (const d of findDialogs()) {
      const isErr = ERR.test(d.text);
      const c = chooseButton(d.btns, isErr, d.aria);
      if (!c) continue;
      const last = clickedAt.get(c.el) || 0;
      if (performance.now() - last < 700) continue;
      clickedAt.set(c.el, performance.now());
      try { pressCenter(c.el); } catch (_) {}
      log({ dialog: 'cleared', clicked: c.text || 'close', text: norm(d.text).slice(0, 44) });
      if (phase === 'reserve' || phase === 'done') {
        if (ORPHAN.test(d.text)) fixOrphan(); else judgeAfterDialog(isErr);
      }
      return;
    }
  }
  function judgeAfterDialog(isErr) {
    const my = epoch;
    setTimeout(async () => {
      if (my !== epoch || !(phase === 'reserve' || phase === 'done') || !rects.length) return;
      let states = [];
      try { states = lastT.map(seatState); } catch (_) {}
      const n = states.length, kept = states.filter((x) => x === 'selected').length;
      if (n && kept === n) {
        if (isErr) {
          attempts++;
          log({ retry: attempts, why: 'same seats', seats: n });
          phase = 'reserve'; reserveAt = performance.now();
          reserveFast(my, { reserved: false }, performance.now());
        } else log({ waiting: 'checkout', note: 'confirmation accepted' });
        return;
      }
      if (!n) { if (isErr) rearm('dialog'); return; }
      for (let i = 0; i < lastT.length; i++) {
        if (states[i] !== 'selected') continue;
        const el = findAnyAt(lastT[i]);
        if (el) { try { pressCenter(el); } catch (_) {} }
      }
      if (kept) await settle(20, 160);
      if (my !== epoch) return;
      rearm('seats released');
    }, 250);
  }
  let dmo = null, dTimer = null;
  function startDialogWatch() {
    stopDialogWatch();
    dmo = new MutationObserver(() => handleDialogs());
    dmo.observe(document.documentElement, { subtree: true, childList: true, attributes: true });
    dTimer = setInterval(handleDialogs, 50);
  }
  function stopDialogWatch() { if (dmo) { dmo.disconnect(); dmo = null; } if (dTimer) { clearInterval(dTimer); dTimer = null; } }

  function rearm(why) {
    if (!rects.length || !engineOn()) return;
    epoch++; attempts++;
    orphanTries = 0;
    armed = true; done = false; scanning = false; phase = 'scan';
    boxEls.forEach((b) => { b.className = 'box armed'; });
    log({ retry: attempts, why });
    startWatch();
    scan();
  }

  }, 120);

    else if (phase === 'reserve') {

    }
